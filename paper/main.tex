\begin{abstract}
We study finite-window recognition of the canonical orientation of the fixed
rank-four $q=3$ Demu\v{s}kin pro-$3$ group. Let $P_n$ denote its Zassenhaus
filtration and put $Q_k=G/P_{3^{k-1}+1}$. For a principal-unit character
$\rho:Q_k\to U_{1,k}:=1+3(\mathbb Z/3^k\mathbb Z)$, let
$\mathsf K_k(Q_k,\rho)$ mean that
\[
H^1(Q_k,\mathbb Z/3^k(\rho))\longrightarrow H^1(Q_k,\mathbb F_3)
\]
is surjective. We prove
\[
\mathsf K_k(Q_k,\rho)\Longleftrightarrow
\rho=\chi_G\pmod{3^k}
\]
for every $k\ge2$. The canonical orientation and its Kummerian uniqueness
are classical. The finite contribution is the arbitrary-candidate
factorization through the bare finite quotient, the fixed-scope selector
minimality theorem, and the intrinsic one-dimensional cup-line carrier for
the surviving first-order variation. Publication novelty is stated
conditionally; no absolute priority claim is made.
\end{abstract}

\section{Introduction}

Let
\[
G=\langle x_1,x_2,x_3,x_4\mid
x_1^3[x_1,x_2][x_3,x_4]=1\rangle,
\qquad [a,b]=a^{-1}b^{-1}ab.
\]
Write $P_n=P_n(G)$ for the Zassenhaus filtration. The canonical
Demu\v{s}kin orientation is a continuous homomorphism
\[
\chi_G:G\longrightarrow 1+3\mathbb Z_3.
\]
Throughout we fix the convention
\[
\chi_G(x_2)=(1-3)^{-1},
\]
so in particular $\chi_G(x_2)\equiv4\pmod9$. This is only a
normalization convention; all intrinsic statements below are independent of
this presentation coordinate.
For an orientation $\theta$, the coefficient module
$\mathbb Z_3(\theta)$ is the usual rank-one twisted module.

The Kummerian condition says that the coefficient-reduction maps
\[
H^1(G,\mathbb Z_3(\theta)/3^n)
\longrightarrow H^1(G,\mathbb F_3)
\]
are surjective for all $n$. The existence and uniqueness of the
Kummerian orientation for Demu\v{s}kin groups are classical, due to
Labute and the later Kummerian/1-cyclotomic theory.

The question here is narrower and finite:
can the reduction $\chi_G\bmod 3^k$ be recognized from the bare finite
Zassenhaus quotient
\[
Q_k=G/P_{3^{k-1}+1}
\]
without supplying $q$, a presentation coordinate, or the canonical
orientation itself?

For a principal-unit character $\rho:Q_k\to U_{1,k}$ define
\[
\Kk(Q_k,\rho):
\quad
H^1(Q_k,\Ak(\rho))
\longrightarrow H^1(Q_k,\mathbb F_3)
\quad\text{is surjective}.
\]

\begin{theorem}[Finite-window recognition]\label{thm:finite-window-recognition}
For every $k\ge2$ and every principal-unit
$\rho:Q_k\to U_{1,k}$,
\[
\boxed{
\Kk(Q_k,\rho)\Longleftrightarrow
\rho=\chi_G\pmod{3^k}.
}
\]
\end{theorem}

The proof is organized as follows. First, an explicit
Jennings--Lazard calculation gives the exact finite depth needed to
factor every arbitrary twisted crossed cocycle. Second, a finite
one-relator Fox criterion reduces the base case to four equations.
Third, a coefficient-variation identity together with $PD^2$ duality
gives intrinsic uniqueness at successive $3$-adic levels. Finally, an
independent lower-bound argument proves the exact selector depth
\[
n_{\mathrm{selector}}(k)=3^{k-1}+1.
\]
We then identify the intrinsic one-dimensional cup-product image in the
finite quotient and explain precisely what it does and does not carry.

\section{The finite window and arbitrary-candidate factorization}

Put
\[
A_k=\mathbb Z/3^k\mathbb Z,\qquad
U_j=1+3^jA_k\quad(j\ge1),
\]
with $U_j=1$ for $j\ge k$, and
\[
S_k=A_k\rtimes U_1,\qquad
(a,u)(b,v)=(a+ub,uv).
\]
Let $P_n(S_k)$ denote the $3$-Zassenhaus filtration.

\begin{lemma}[U1: finite-depth Zassenhaus filtration]
For $n\ge1$, put
\[
e(n)=\lceil\log_3 n\rceil,\qquad e(1)=0.
\]
Then
\[
\boxed{
P_n(S_k)=3^{e(n)}A_k\rtimes U_{e(n)+1}.
}
\]
In particular,
\[
P_{3^{k-1}}(S_k)=3^{k-1}A_k\ne1,
\qquad
P_{3^{k-1}+1}(S_k)=1.
\]
\end{lemma}

\begin{proof}
Use the Jennings--Lazard description
\[
P_n(S_k)=\prod_{i3^j\ge n}\gamma_i(S_k)^{3^j}.
\]
A direct commutator calculation gives
\[
\gamma_2(S_k)=3A_k,\qquad
\gamma_i(S_k)=3^{i-1}A_k\quad(i\ge2).
\]
Indeed, $[u,a]=(u-1)a$ for $u\in U_1$, $a\in A_k$, and
$u=4$ already generates $3A_k$.

For $j\ge0$,
\[
S_k^{3^j}=3^jA_k\rtimes U_{j+1}.
\]
The translation part is divisible by $3^j$ because
\[
(a,1)^{3^j}=(3^ja,1),
\]
and the unit part satisfies
\[
U_1^{3^j}=U_{j+1}
\]
by the usual $3$-adic valuation of
$(1+3^sb)^{3^j}-1$.

Let $e=e(n)$. The $i=1$ term with $3^j\ge n$ first occurs at
$j=e$ and contributes
\[
3^eA_k\rtimes U_{e+1}.
\]
If $i\ge2$ and $i3^j\ge n$, then
$i-1+j\ge e$, so
\[
\gamma_i(S_k)^{3^j}
=3^{i-1+j}A_k
\subseteq3^eA_k.
\]
Thus no further factor enlarges the subgroup, proving the formula.
At $n=3^{k-1}$ one has $e=k-1$, while at
$n=3^{k-1}+1$ one has $e=k$.
\end{proof}

\begin{lemma}[U2: arbitrary-candidate factorization]
Let $H$ be any pro-$3$ group, let
$\rho:H\to U_{1,k}$ be a continuous character, and let
$z\in Z^1(H,A_k(\rho))$. Then both $z$ and $\rho$ kill
$P_{3^{k-1}+1}(H)$. Hence they factor through
\[
Q_k(H)=H/P_{3^{k-1}+1}(H).
\]
Because $\rho\equiv1\pmod3$, the kernel $P_{3^{k-1}+1}(H)$ is contained in
$P_2(H)=\Phi(H)$ and acts trivially on $A_k(\rho)$. Thus inflation gives
natural isomorphisms
\[
H^1(Q_k(H),A_k(\bar\rho))
\xrightarrow{\sim}
H^1(H,A_k(\rho))
\]
and
\[
H^1(Q_k(H),\mathbb F_3)
\xrightarrow{\sim}
H^1(H,\mathbb F_3),
\]
and these identifications commute with coefficient reduction.
\end{lemma}

\begin{proof}
The crossed-cocycle identity is equivalent to
\[
g\longmapsto(z(g),\rho(g))
\]
being a homomorphism $H\to S_k$. Lemma U1 and functoriality of the
Zassenhaus filtration therefore show that
$P_{3^{k-1}+1}(H)$ is killed by both components.

The same argument for coboundaries gives the twisted $H^1$ isomorphism.
For trivial coefficients,
\[
P_{3^{k-1}+1}(H)\subseteq P_2(H)=\Phi(H),
\]
so every homomorphism to $\mathbb F_3$ kills the kernel and inflation is
an isomorphism on $H^1$. Naturality of coefficient reduction gives the
commutative square.
\end{proof}

\section{The finite Kummer/Fox criterion}

Let $F$ be free pro-$3$ on $x_1,\ldots,x_d$ and let
\[
H=F/\overline{\langle\!\langle r\rangle\!\rangle}
\]
be a minimal one-relator quotient, where $F$ is free pro-$3$ of rank
$d=\dim_{\mathbb F_3}H^1(H,\mathbb F_3)$ and $r\in\Phi(F)$. Since $r\in\Phi(F)$, the relator imposes no
linear relation on $F/\Phi(F)$, so
\[
H/\Phi(H)\cong F/\Phi(F),\qquad
\dim_{\mathbb F_3}H^1(H,\mathbb F_3)=\operatorname{rank}F=d.
\]
Thus the displayed value of $d$ is the rank of the free group as well as
the minimal number of generators of $H$. For a principal-unit character
$\rho:H\to U_{1,k}$, write $F_i(\rho)\in A_k$ for the twisted Fox
coefficients, normalized by
\[
z(r)=\sum_iF_i(\rho)z(x_i).
\]

\begin{proposition}[U3: finite Kummer/Fox criterion]
Under the minimal one-relator presentation specified above,
\[
\boxed{
\mathsf K_k(H,\rho)\Longleftrightarrow
F_i(\rho)=0\quad(1\le i\le d).
}
\]
\end{proposition}

\begin{proof}
A crossed cocycle is determined on the free generators, and the single
relation is exactly
\[
\sum_iF_i(\rho)a_i=0.
\]
Because $\rho\equiv1\pmod3$, the coefficient action is trivial after
reduction modulo $3$, and minimality identifies
$H^1(H,\mathbb F_3)$ with $\mathbb F_3^d$.

If the reduction map is surjective, choose a lift with
$a_i\equiv1\pmod3$ and $a_j\equiv0\pmod3$ for $j\ne i$.
The following divisibility induction is carried out simultaneously for all
$i$. If all $F_j\in3^mA_k$, then
\[
0=F_i(1+3b_i)+\sum_{j\ne i}F_j(3b_j)
=F_i+3\sum_jF_jb_j,
\]
so $F_i\in3^{m+1}A_k$. Starting at $m=0$ and iterating gives
$F_i=0$ for every $i$. Conversely, if all $F_i=0$, every generator-value vector
in $A_k^d$ satisfies the relation, so every mod-$3$ class lifts.
\end{proof}

\section{The base case $k=2$}

For
\[
r=x_1^3[x_1,x_2][x_3,x_4],
\]
write $a_i=\rho_2(x_i)\in\{1,4,7\}\subset\mathbb Z/9\mathbb Z$.
For the convention
\[
z(gh)=z(g)+\rho_2(g)z(h),
\]
the twisted Fox coefficients are
\[
\begin{aligned}
F_1&=\frac{a_1^2+a_1a_2+a_2}{a_2},
&
F_2&=\frac{a_1^2(a_1-1)}{a_2},
\\
F_3&=-\frac{a_1^3(a_4-1)}{a_3a_4},
&
F_4&=\frac{a_1^3(a_3-1)}{a_3a_4}.
\end{aligned}
\]
All denominators are units modulo $9$. By U3, the finite Kummer predicate
is equivalent to the vanishing of all four coefficients. Hence
\[
F_2=0\Rightarrow a_1=1,\qquad
F_3=F_4=0\Rightarrow a_3=a_4=1,
\]
and then
\[
F_1=0
\Longleftrightarrow
1+2a_2=0\pmod9
\Longleftrightarrow
a_2=4.
\]
Thus the unique level-$2$ candidate is
\[
\boxed{\rho_2=(1,4,1,1)=\chi_G\bmod9.}
\]
The surviving candidate is indeed Kummerian because it is the reduction
of the classical canonical Kummerian orientation. This establishes the
base step without using the desired recognition theorem as an input.

\section{Intrinsic uniqueness at higher levels}

\begin{lemma}[U5a: reduction]
If $\mathsf K_k(G,\rho_k)$ holds, then
$\mathsf K_{k-1}(G,\rho_{k-1})$ holds after reduction modulo $3^{k-1}$.
\end{lemma}

\begin{proof}
Reduce a level-$k$ crossed cocycle modulo $3^{k-1}$.
A lift of every class in $H^1(G,\mathbb F_3)$ at level $k$ therefore
gives a lift at level $k-1$.
\end{proof}

\begin{lemma}[U5b: coefficient-extension variation]
Suppose $\rho_k$ and $\rho_k'$ lift the same $\rho_{k-1}$ and
\[
\rho_k'=\rho_k(1+3^{k-1}\nu),
\qquad \nu\in Z^1(G,\mathbb F_3).
\]
Let
\[
\delta_{\rho_k},\delta_{\rho_k'}:
H^1(G,\mathbb F_3)
\longrightarrow
H^2(G,A_{k-1}(\rho_{k-1}))
\]
be the connecting maps of the corresponding coefficient extensions.
Then
\[
\boxed{
\delta_{\rho_k'}-\delta_{\rho_k}
=
\iota_*\circ(\nu\smile-),
}
\]
where
\[
\iota:\mathbb F_3\hookrightarrow A_{k-1}(\rho_{k-1}),
\qquad
\iota(1)=3^{k-2}.
\]
\end{lemma}

\begin{proof}
Use
\[
0\to A_{k-1}(\rho_{k-1})
\xrightarrow{j}A_k(\rho_k)
\xrightarrow{\pi}\mathbb F_3\to0,
\qquad j(a)=3a,
\]
and the common set-theoretic section $s(1)=1$.
For $c\in Z^1(G,\mathbb F_3)$,
\[
d_sc(g,h)=g\!\cdot s(c(h))-s(c(gh))+s(c(g)).
\]
The two coefficient actions differ by
\[
g\!\cdot' s(c(h))-g\!\cdot s(c(h))
=3^{k-1}\nu(g)c(h).
\]
Hence
\[
d_s'c-d_sc
=
3^{k-1}(\nu\smile c)
=
j\!\left(3^{k-2}(\nu\smile c)\right).
\]
Passing to cohomology gives the asserted identity.
\end{proof}

\begin{lemma}[U5c: $PD^2$ socle injectivity]
For the canonical orientation $\chi$, the inclusion
\[
\iota:\mathbb F_3\hookrightarrow A_{k-1}(\chi_{k-1}),
\qquad 1\mapsto3^{k-2},
\]
induces an injection
\[
\iota_*:
H^2(G,\mathbb F_3)
\hookrightarrow
H^2(G,A_{k-1}(\chi_{k-1})).
\]
\end{lemma}

\begin{proof}
Let $M^\vee=\operatorname{Hom}(M,\mathbb Q_3/\mathbb Z_3)$ and
let $\mathcal I=\mathbb Q_3/\mathbb Z_3(\chi)$ be the dualizing module,
where $g$ acts on $\mathbb Q_3/\mathbb Z_3$ in the twist by
multiplication by $\chi(g)$. Thus, for
$M=A_{k-1}(\chi_{k-1})$, the contragredient twist in $M^\vee$
cancels the $\chi$-twist in $\mathcal I$. Finite-coefficient $PD^2$
duality gives the natural isomorphism
\[
H^2(G,M)^\vee
\cong
H^0(G,M^\vee\otimes\mathcal I)
\]
for finite $3$-primary modules $M$. Consequently
\[
H^2(G,A_{k-1}(\chi_{k-1}))^\vee\cong A_{k-1},
\qquad
H^2(G,\mathbb F_3)^\vee\cong\mathbb F_3.
\]
The naturality of $PD^2$ duality identifies the dual of the inclusion
$\iota:\mathbb F_3\hookrightarrow A_{k-1}(\chi_{k-1})$,
$1\mapsto3^{k-2}$, with the induced map on the invariant duals
\[
A_{k-1}\longrightarrow\mathbb F_3,
\qquad a\longmapsto a\bmod3.
\]
Equivalently, the corresponding duality square commutes with coefficient
reduction. This map is surjective, hence its Pontryagin dual
$\iota_*$ is injective.
\end{proof}

\begin{proposition}[U5: intrinsic uniqueness]
If $\rho_k$ and $\rho_k'$ both satisfy $\mathsf K_k$ and both reduce to
$\chi_{k-1}$, then
\[
\rho_k=\rho_k'.
\]
\end{proposition}

\begin{proof}
Surjectivity of the coefficient-reduction map is equivalent to the
vanishing of its connecting map. Hence
\[
0=\delta_{\rho_k'}-\delta_{\rho_k}
=\iota_*(\nu\smile-).
\]
By U5c,
\[
\nu\smile v=0
\qquad\text{for all }v\in H^1(G,\mathbb F_3).
\]
The Demu\v{s}kin cup pairing is nondegenerate, so $\nu=0$.
\end{proof}

\section{Proof of the finite-window recognition theorem}

The canonical orientation $\chi_G$ is Kummerian by the classical
Demu\v{s}kin theory; Labute's Theorem~4 gives the existence and uniqueness
of this orientation and its finite crossed-derivation criterion. Thus
$\chi_G\bmod3^k$ satisfies the finite predicate. Lemma~U2 transfers this
statement between $G$ and $Q_k$ in both directions, so
\[
\mathsf K_k(Q_k,\rho_k)\Longleftrightarrow
\mathsf K_k(G,\rho_k).
\]

Conversely, suppose $\mathsf K_k(Q_k,\rho_k)$ holds. Lemma U2 transfers
the predicate to $G$, and U5a transfers it to every lower level.
At level $2$, the explicit base calculation gives
\[
\rho_2=\chi_G\bmod9.
\]
For $j\ge3$, $\rho_j$ and $\chi_G\bmod3^j$ have the same reduction to
level $j-1$, and both satisfy the level-$j$ predicate. U5 therefore gives
\[
\rho_j=\chi_G\bmod3^j.
\]
Induction yields
\[
\rho_k=\chi_G\bmod3^k.
\]
This proves the theorem.

\section{The intrinsic one-dimensional cup carrier}

Define
\[
\boxed{
C_k=
\operatorname{im}\!\left(
H^1(Q_k,\mathbb F_3)\otimes H^1(Q_k,\mathbb F_3)
\xrightarrow{\cup}
H^2(Q_k,\mathbb F_3)
\right).
}
\]

\begin{proposition}[D3: finite cup-line]
For every $k\ge2$,
\[
\boxed{\dim_{\mathbb F_3}C_k=1,}
\]
and inflation $C_k\to H^2(G,\mathbb F_3)$ is injective.
\end{proposition}

\begin{proof}
Write $G=F/R$ with $F$ free pro-$3$ on four generators and $R$
normally generated by $r$. Put
\[
n=3^{k-1}+1,
\qquad
Q_k=F/R_k,
\qquad
R_k=R\,P_n(F).
\]
Since $n\ge3$, $P_n(F)\subseteq P_3(F)$. Thus
\[
\operatorname{im}\bigl(R_k\to P_2(F)/P_3(F)\bigr)
=
\operatorname{im}\bigl(R\to P_2(F)/P_3(F)\bigr).
\]
Because $R=\overline{\langle\!\langle r\rangle\!\rangle}$ and conjugation,
$3$-powering, and multiplication by elements of $P_3(F)$ do not change
the degree-two initial form, this latter image is exactly
\[
\mathbf F_3\cdot\operatorname{in}_2(r),
\qquad
\operatorname{in}_2(r)=[X_1,X_2]+[X_3,X_4].
\]
Also $R_k\subseteq\Phi(F)$, so this is a minimal pro-$3$ presentation.

For a minimal pro-$p$ presentation, Proposition~7.1 of
Miná\v{c}--Pasini--Quadrelli--T\^an gives a commutative pairing diagram
identifying evaluation of cup products on defining relations with the
degree-two initial-form map
\[
R_k/R_k^p[R_k,F]\longrightarrow P_2(F)/P_3(F).
\]
Applied to $F/R_k$, its image is exactly the one-dimensional span of
$\operatorname{in}_2(r)$, because $P_n(F)\subseteq P_3(F)$ for
$n=3^{k-1}+1\ge3$. Hence the dual finite cup map has rank one, and so does
the cup-product map itself. This is a finite-presentation argument; it does
not use injectivity of $H^2(Q_k)\to H^2(G)$.

The same relation-module/cup pairing identifies the unique nonzero
finite cup line with the functional dual to the quadratic initial form
$\operatorname{in}_2(r)$. After inflation to $G$, this is exactly the
nonzero Demu\v{s}kin cup class dual to the defining relator. Hence the
inflation of $C_k$ is nonzero. Since both $C_k$ and
$H^2(G,\mathbb F_3)$ are one-dimensional, the inflation is injective.
\end{proof}

The coefficient connecting map itself has target
\[
H^2(Q_k,A_{k-1}(\rho_{k-1})),
\]
not $H^2(Q_k,\mathbb F_3)$. Thus the statement about $C_k$ is not that the
entire connecting map has codomain $C_k$. Rather, $C_k$ is the intrinsic
finite carrier of the surviving first-order cup variation.

Indeed, on the canonical lower-level branch write
\[
\rho_k=\chi_k(1+3^{k-1}\nu).
\]
U5b gives, after inflation to $G$,
\[
\delta_{\rho_k}(f)-\delta_{\chi_k}(f)
=
\iota_*(\nu\smile\bar f).
\]
The canonical connecting map vanishes. For every $\nu\ne0$, cup
nondegeneracy gives $f$ with
\[
\nu\smile\bar f\ne0.
\]
By definition, $C_k$ is the image of the finite cup-product map, so
$\nu\smile\bar f\in C_k$. After applying the coefficient inclusion
$\iota_*$, the same class is represented by the corresponding cup-product
variation in $H^2(G,\mathbf F_3)$; by the inflation injectivity just proved,
its finite representative is nonzero. Thus the variation is detected by the
one-dimensional finite cup line. Hence $C_k$ supplies a one-dimensional
linear detector for every false next-digit candidate.

\begin{remark}
This is minimal only in the declared linear selector-carrier category:
a zero-dimensional vector space cannot detect any false candidate, while
$C_k$ has dimension one. No absolute minimality among arbitrary nonlinear
finite observations is claimed.
\end{remark}

\section{Exact selector depth}

For $m\ge1$, write
\[
W_m=G/P_m.
\]
Here, for a pro-$3$ group, $G^{3^e}$ denotes the closed subgroup generated
by all $3^e$-th powers; the standard Zassenhaus formula gives
$G^{3^e}\subseteq P_{3^e}(G)$.
For each $m$ and $k\ge2$, define the domain
\[
\mathcal D_{k,m}
=
\operatorname{Hom}_{\mathrm{cont}}(W_m,U_{1,k}),
\qquad
W_m=G/P_m,
\]
and, for $\rho\in\mathcal D_{k,m}$, define the depth-$m$ selector by
\[
\mathsf K_{k,m}(W_m,\rho):
\quad
H^1(W_m,A_k(\rho))
\longrightarrow H^1(W_m,\mathbb F_3)
\quad\text{is surjective}.
\]
Thus the predicate is defined only for characters that actually factor
through $W_m$. In particular, when $m\le3^{k-2}$ the target character
$\chi_G\bmod3^k$ does not belong to $\mathcal D_{k,m}$, by the
shallow-range calculation below. The lower-bound argument therefore has two
separate parts: below $3^{k-1}$ the target is not even in the selector
domain, while for $3^{k-2}<m\le3^{k-1}$ the target is in the domain but
the selector fails.

\begin{proposition}[D4: exact selector depth]
For the fixed rank-four $q=3$ Demu\v{s}kin group, the least Zassenhaus
index $m$ for which the finite Kummer predicate recognizes
$\chi_G\bmod3^k$ is
\[
\boxed{
n_{\mathrm{selector}}(k)=3^{k-1}+1.
}
\]
\end{proposition}

\begin{proof}
Put $N=3^{k-1}$.

If $m\le3^{k-2}$, put $e=\lceil\log_3m\rceil\le k-2$.
Since $x_2^{3^e}\in G^{3^e}\subseteq P_{3^e}(G)\subseteq P_m(G)$,
while
\[
v_3\!\left(\chi_G(x_2)^{3^e}-1\right)
=v_3\!\left((1-3)^{-3^e}-1\right)=1+e<k,
\]
the canonical character does not factor through $W_m$.

Now suppose
\[
3^{k-2}<m\le N.
\]
Then $\chi_G\bmod3^k$ does factor through $W_m$, but the image of $x_2$
has exact order $N$. Put
\[
u=\chi_G(x_2)=(1-3)^{-1}.
\]
Take the mod-$3$ character with value $1$ on $x_2$ and $0$ on the other
generators. Any twisted lift $z$ would satisfy
$z(x_2)\equiv1\pmod3$, while $x_2^N=1$ in $W_m$ forces
\[
0=z(x_2^N)=S_N(u)z(x_2),
\qquad
S_N(u)=1+u+\cdots+u^{N-1}.
\]
Here $v_3(u-1)=1$, not $0$. LTE gives
\[
v_3(u^N-1)=v_3(u-1)+v_3(N)=1+(k-1)=k,
\]
and therefore
\[
v_3(S_N(u))
=v_3(u^N-1)-v_3(u-1)=k-1.
\]
Thus $S_N(u)$ is nonzero modulo $3^k$ but divisible by $3$, and the
relation forces $z(x_2)\equiv0\pmod3$, a contradiction. Hence the
Kummer predicate fails throughout this range.

The upper bound $m=N+1$ is Theorem~\ref{thm:finite-window-recognition} proved above. Therefore the threshold is exact.
\end{proof}

\begin{remark}
This is selector minimality for the fixed rank-four $q=3$ category and the
specific finite Kummer selector. It is independent of the affine
representation sharpness theorem. It is not an absolute lower bound for
arbitrary nonlinear finite observations.
\end{remark}

\section{Detailed verification of the finite-depth calculation}

Put $A_k=\mathbb Z/3^k\mathbb Z$ and $U_s=1+3^sA_k$.
For $u=1+3^sb$, $s\ge1$,
\[
(1+3^sb)^{3^j}-1
=
3^{s+j}b+\sum_{m\ge2}\binom{3^j}{m}3^{sm}b^m.
\]
The first term has valuation $s+j$ whenever $s+j<k$, while the remaining
terms have strictly larger valuation. Thus
\[
(1+3^sb)^{3^j}\equiv1+3^{s+j}b\pmod{3^k},
\]
and the power is $1$ when $s+j\ge k$. Hence
\[
U_1^{3^j}=U_{j+1}.
\]
Together with
\[
(a,1)^{3^j}=(3^ja,1)
\]
this yields
\[
S_k^{3^j}=3^jA_k\rtimes U_{j+1}.
\]
For $i\ge2$,
\[
\gamma_i(S_k)=3^{i-1}A_k.
\]
The Jennings--Lazard product (in the standard Zassenhaus/Jennings framework~\cite{MinacRogelstadTan2016}) then gives
\[
P_n(S_k)=3^{\lceil\log_3n\rceil}A_k
\rtimes U_{\lceil\log_3n\rceil+1}.
\]
In particular,
\[
P_{3^{k-1}}(S_k)=3^{k-1}A_k\ne1,
\qquad
P_{3^{k-1}+1}(S_k)=1.
\]

\section{Cochain-level verification of coefficient variation}

Use
\[
0\to A_{k-1}(\rho_{k-1})
\xrightarrow{j}A_k(\rho_k)
\xrightarrow{\pi}\mathbb F_3\to0,
\qquad j(a)=3a.
\]
For a common section $s(1)=1$ and
$c\in Z^1(G,\mathbb F_3)$,
\[
d_sc(g,h)=g\!\cdot s(c(h))-s(c(gh))+s(c(g)).
\]
If
\[
\rho_k'=\rho_k(1+3^{k-1}\nu),
\]
then
\[
d_s'c-d_sc=3^{k-1}(\nu\smile c)
=j\!\left(3^{k-2}(\nu\smile c)\right).
\]
Thus
\[
\delta_{\rho_k'}-\delta_{\rho_k}
=
\iota_*\circ(\nu\smile-).
\]
The sign is fixed by
\[
(\nu\smile c)(g,h)=\nu(g)c(h).
\]

\section{Derivation of the twisted Fox coefficients}\label{app:fox-derivation}

Let $a_i=\rho(x_i)$ and $X_i=z(x_i)$ for a crossed cocycle
$z$ with convention $z(gh)=z(g)+\rho(g)z(h)$. Then
\[
z(x_i^3)=(1+a_i+a_i^2)X_i.
\]
For $a=\rho(x)$, $b=\rho(y)$, $X=z(x)$ and $Y=z(y)$, direct use of
$z(g^{-1})=-\rho(g)^{-1}z(g)$ gives, for
$[x,y]=x^{-1}y^{-1}xy$,
\[
z([x,y])=\frac{1-b}{ab}X+\frac{a-1}{ab}Y.
\]
Applying this first to $[x_1,x_2]$ and then to $[x_3,x_4]$, and observing
that every commutator has trivial image under the one-dimensional
character, yields
\[
\begin{aligned}
F_1&=(1+a_1+a_1^2)+a_1^3\frac{1-a_2}{a_1a_2}
=\frac{a_1^2+a_1a_2+a_2}{a_2},\\
F_2&=a_1^3\frac{a_1-1}{a_1a_2}
=\frac{a_1^2(a_1-1)}{a_2},\\
F_3&=a_1^3\frac{1-a_4}{a_3a_4},\\
F_4&=a_1^3\frac{a_3-1}{a_3a_4}.
\end{aligned}
\]
All denominators are units in $A_k$ because $\rho$ is principal-unit valued.
This derives the coefficients used in the level-$2$ calculation rather than
treating them as an unproved symbolic input.
